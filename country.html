async function guardarCambiosEnSupabase() { const statusDiv = document.getElementById('status'); const keys = Object.keys(window.cambiosPendientes); if (keys.length === 0) { statusDiv.textContent = "⚠️ No hay cambios pendientes para guardar."; statusDiv.style.color = '#e08a1e'; return; } statusDiv.textContent = "⏳ Guardando nuevos registros en Supabase..."; statusDiv.style.color = '#00d2ff'; try { const today = new Date().toISOString().split('T')[0]; for (let key of keys) { let [year, country] = key.split('|'); let position = window.cambiosPendientes[key]; if (position === '') position = null; // Usamos UPSERT indicando que si colisiona la combinación country,year haga un merge/update const response = await fetch(`${SUPABASE_URL}/country_rankings`, { method: 'POST', headers: { 'apikey': SUPABASE_ANON_KEY, 'Authorization': 'Bearer ' + SUPABASE_ANON_KEY, 'Content-Type': 'application/json', 'Prefer': 'resolution=merge-duplicates, return=minimal' }, body: JSON.stringify({ country: country, year: parseInt(year), position: position, date: today }) }); if (!response.ok) { const errDetail = await response.text(); throw new Error(`Error al guardar registro para ${country} en ${year}: ${errDetail}`); } window.valoresOriginales[key] = position; window.fechasActualizacion[country] = today; } window.cambiosPendientes = {}; renderizarTablaCompleta(); actualizarEstadoBotonGuardar(); statusDiv.textContent = "✅ ¡Registros guardados con éxito en country_rankings!"; statusDiv.style.color = '#3fb950'; } catch (err) { statusDiv.textContent = "❌ " + err.message; statusDiv.style.color = '#f85149'; console.error(err); } }
